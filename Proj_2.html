#Part 1A: Exponential Transformation

Exp <- function(x, k) {
  fun1A <- 1  #first term
  for (i in 1:k) {        #create for loop
    fun1A <- fun1A + (x^i) / factorial(i)
  }
  
  return(fun1A)
}

#Part 1B: Sample mean and sample sd

# Function to calculate the sample mean
sample_mean <- function(x) {
  N <- length(x)  # Get the number of observations
  sum_x <- 0  # Initialize the sum
  for (i in 1:N) {
    sum_x <- sum_x + x[i]  # Add each element to the sum
  }
  return(sum_x / N)  # Return the average
}

# Function to calculate the sample standard deviation
sample_sd <- function(x) {
  N <- length(x)  # Get the number of observations
  mean_x <- sample_mean(x)  # Compute the sample mean
  
  sum_squared_diff <- 0  # Initialize the sum of squared differences
  for (i in 1:N) {
    sum_squared_diff <- sum_squared_diff + (x[i] - mean_x)^2  # Compute the squared differences
  }
  
  return(sqrt(sum_squared_diff / (N - 1)))  # Return the sample standard deviation
}

# Part 1C: Confidence Intervals
# Assuming sample_mean and sample_sd functions already exist

calculate_CI <- function(x, conf = 0.95) {
  N <- length(x)  # Get the number of observations
  mean_x <- sample_mean(x)  # sample mean
  sd_x <- sample_sd(x)  # sample sd
  SE <- sd_x / sqrt(N)  # Standard error of the mean
  
  alpha <- 1 - conf  # alpha
  degrees_freedom <- N - 1  # Df for the t-distribution
  t_score <- qt(p = alpha / 2, df = degrees_freedom, lower.tail = FALSE)  # t-score
  
  # confidence interval
  margin_of_error <- t_score * SE
  lower_bound <- mean_x - margin_of_error
  upper_bound <- mean_x + margin_of_error
  
  # Return the confidence interval as a named vector
  return(c(lower_bound = lower_bound, upper_bound = upper_bound))
}

#3B Create Function to plot rainfall distribution for a given city and year
ggplot_rainfall <- function(city_name, year) {
  #1.Check if city_name and year exist in the dataset
  if (!(city_name %in% df$city_name)) {
    stop("Error in 'if()': This city does not exist in the dataset.")
  }
  if (!(year %in% df$year)) {
    stop("Error in 'if()': This year does not exist in the dataset.")
  }
  
  #2. Filter data based on city_name and year
  filtered_rainfall <- df %>%
    filter(city_name == city_name, year == year)
  
  #3. Plot histogram of log-transformed rainfall
  ggplot(filtered_rainfall, aes(x = log(rainfall))) +
    geom_histogram(binwidth= 0.5, fill = "lightblue", color = "black") +
    labs(
      title = paste("Rainfall Distribution in", city_name, "for the Year", year),
      subtitle = paste("Log transformed rainfall distribution for", city_name, "in", year),
      x = "Log(Rainfall)",
      y = "Frequency",
      caption = "Data source: TidyTuesday rainfall data, Jan 2020"
    ) +
    theme_minimal() +
    theme(
      plot.title = element_text(size = 16, face = "bold"),
      plot.subtitle = element_text(size = 12),
      axis.text = element_text(size = 10),
      axis.title = element_text(size = 12)
    )
}

I named the function starting with ggplot so the user recognizes that this function creates a plot and uses the ggplot package. First, an ‘if()’ was used to check if the city and year of interest were part of the dataset. If the city or year is not included, the corresponding error message reports where the error came from and why.This checkpoint alerts the user on the first step and requires a solution before moving forward. This prevents trying to determine the source of error later. Next, the dataset is filtered to only include that city and year. This is a necessary step prior to plotting to ensure the desired output. Lastly, I specified binwidth to override the default of 30 bins and added informative labels, text sizes, and colors to improve the plot.
